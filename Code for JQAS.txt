##################### Tournament designs

generate_R1 <- function(r, n){

  if(r == 1){

    ## Random draw
    R1 <- sample(1:n, replace = FALSE)

  } else if(r == 2){

    ## Ordered draw
    R1 <- 1:n

  } else if(r == 3){

    ## Randomly positioned consecutive pairs
    pairs <- sample(seq(1, n-1, 2), replace = FALSE)
    R1 <- insert(pairs,
                 ats = 1:(n/2),
                 values = pairs + 1)

  } else if(r == 4){

    ## Fixed halves (Seeds 1 & 2)
    R1 <- c(1, sample(3:n, replace = FALSE), 2)

  } else if(r == 5){

    ## Fixed quarters (Seeds 1-4, fixed sum)
    R1 <- c(1, sample(5:n, replace = FALSE), 2)
    R1 <- append(R1,
                 c(4,3),
                 (n-2)/2)

  } else if(r == 6){

    ## Fixed quarters (Seeds 1-4, fixed difference)
    R1 <- c(1, sample(5:n, replace = FALSE), 2)
    R1 <- append(R1,
                 c(3,4),
                 (n-2)/2)

  } else if(r == 7){

    ## Fixed eighths (Top 8, fixed sum)
    R1 <- sample(9:n)
    R1 <- insert(R1,
                 ats = seq(1, n-8, 3),
                 values = c(1,8,5,4,2,7,6,3))

  } else if(r == 8){

    ## Fixed eighths (Top 8, fixed difference)
    R1 <- sample(9:n)
    R1 <- insert(R1,
                 ats = seq(1, n-8, 3),
                 values = c(1,5,7,3,2,6,8,4))

  } else if(r == 9){

    ## Fixed sixteenths (Top 16, fixed sum)
    R1 <- sample(17:n)
    R1 <- insert(R1,
                 ats = seq(1, n-16, 1),
                 values = c(1,16,9,8,5,12,13,4,
                            2,15,10,7,6,11,14,3))

  } else if(r == 10){

    ## Fixed sixteenths (Top 16, fixed difference)
    R1 <- sample(17:n)
    R1 <- insert(R1,
                 ats = seq(1, n-16, 1),
                 values = c(1,9,13,5,7,15,11,3,
                            2,10,14,6,8,16,12,4))

  } else {

    stop("r must be an integer between 1 and 10.")

  }

  return(R1)

}

####################### Competition Kernels/Beat functions


setBeat <- function(B) {
  switch(
    as.character(B),
    "1" = function(a, b) 0.5 + (b - a) * p / (2^k),
    "2" = function(a, b) {
      a <- min(a, 2^k + 1)
      b <- min(b, 2^k + 1)
      0.5 + (b - a) * p / (2^k + 1)
    },
    "3" = function(a, b) 0.5 + (b - a) * p / n,
    "4" = function(a, b) {
      a <- min(a, 2^k + 1)
      b <- min(b, 2^k + 1)
      0.5 + sign(b - a) * (b - a)^2 * p / (2^k + 1)^2
    },
    "5" = function(a, b) 0.5 + sign(b - a) * (b - a)^2 * p / n^2,
    "6" = function(a, b) {
      a <- min(a, 2^k + 1)
      b <- min(b, 2^k + 1)
	1 / (1 + abs(a / b)^alpha)
  },
     "7" = function(a, b) {
      #a <- min(a, 2^k + 1)
      #b <- min(b, 2^k + 1)
  eta <- 0.258 + 0.061 * abs(a-b)
  ifelse(
    (a - b) < 0,
    exp(eta) / (1 + exp(eta)),
    ifelse(
      a == b,
      0.5,
      1 / (1 + exp(eta))
    )
  )
}
)
}

##############  Function for accuracy calculation

compute_accuracy <- function(R_list, n, k) {

  R <- length(R_list)  # number of rounds = log2(n)
  seeds <- 1:(2^k)

  # Expected elimination round for each seed
  # Seed groups: 1, 2, 3:4, 5:8, ..., 2^(k-1)+1 : 2^k
  expected_round <- rep(NA, 2^k)
  start <- 1
  for (g in 1:k) {
    end <- 2^g
    expected_round[start:end] <- R - g + 1
    start <- end + 1
  }
  expected_round[1] <- R  # seed 1 expected to win

  # Actual elimination round for each seed
  actual_round <- rep(NA, 2^k)
  for (s in seeds) {
    for (r in 1:R) {
      if (!(s %in% R_list[[r]])) {
        actual_round[s] <- r - 1
        break
      }
    }
    if (is.na(actual_round[s])) actual_round[s] <- R  # champion
  }

  # Accuracy = proportion of seeds whose actual round matches expected round
  accuracy <- sum(actual_round == expected_round) / (2^k - 1)

  return(accuracy)
}
